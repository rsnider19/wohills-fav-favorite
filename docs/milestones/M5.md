## Goal

Neighbors can enter their house, and the admin can moderate, in the M4 visual language:

- **Enter** (`/enter`): sign in with the phone, snap or pick a photo (resized and stripped of
  EXIF on the device), type the address, drop a pin with GPS or the address lookup and
  fine-tune it on a small dark map, optionally name the house, review the card, send. The
  entry is **pending** until the admin approves it.
- **My entry** (same page once you have one): status (waiting / on the board / not shown),
  the card, and edit or withdraw **until voting opens**; read-only after that. Any change to
  an approved entry sends it back for approval, and the page says so before you save.
- **Moderation** (`/admin`, admin phone only): the waiting queue with photo, address and a
  map link; approve, hide, put back, edit (name, address, pin, photo) and delete hidden
  entries. Approved entries appear in the lineup at once.
- Every error state has copy: location denied, out of area, duplicate address, already
  entered, upload failure, window closed, offline.
- Playwright end-to-end tests for all of it against the local stack, plus unit tests for the
  pure logic. The parade and its Vercel build stay byte-identical.

````ts
/** Pure: the route for a pathname. Unknown paths show the home page. */
export function parseRoute(pathname: string): Route {
  const path = pathname.replace(/\/+$/, '') || '/'
  if (path === PATHS.enter) return 'enter'
  if (path === PATHS.admin) return 'admin'
  return 'home'
}

/** Pure: the href for a route, keeping only the query parameters that steer the whole
 *  app (?preview= for the admin's phase preview). */
export function routeHref(route: Route, search = ''): string {
  const keep = new URLSearchParams()
  const preview = new URLSearchParams(search).get('preview')
  if (preview) keep.set('preview', preview)
  const query = keep.toString()
  return PATHS[route] + (query ? `?${query}` : '')
}

const EVENT = 'lights:navigate'

/** Go to a route without reloading. */
export function navigate(route: Route, { replace = false }: { replace?: boolean } = {}) {
  const href = routeHref(route, window.location.search)
  if (replace) window.history.replaceState(null, '', href)
  else window.history.pushState(null, '', href)
  window.dispatchEvent(new Event(EVENT))
  window.scrollTo({ top: 0 })
}

/** onClick for an <a href={routeHref(...)}>: plain left clicks navigate in place;
 *  modified clicks (new tab, etc.) keep the browser's default. */
export function onRouteClick(route: Route) {
  return (e: MouseEvent<HTMLAnchorElement>) => {
    if (e.defaultPrevented || e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return
    e.preventDefault()
    navigate(route)
  }
}

/** The current route; follows back/forward and navigate(). */
export function useRoute(): Route {
  const [route, setRoute] = useState(() => parseRoute(window.location.pathname))
  useEffect(() => {
    const sync = () => setRoute(parseRoute(window.location.pathname))
    window.addEventListener('popstate', sync)
    window.addEventListener(EVENT, sync)
    return () => {
      window.removeEventListener('popstate', sync)
      window.removeEventListener(EVENT, sync)
    }
  }, [])
  return route
}
````

````ts
import type { LightsSettings } from '../types'

export interface LatLng {
  lat: number
  lng: number
}

export interface Box {
  minLat: number
  maxLat: number
  minLng: number
  maxLng: number
}

/** The contest geofence from lights.settings (the database checks the same box). */
export function geofence(s: Pick<LightsSettings, 'geofence_min_lat' | 'geofence_max_lat' | 'geofence_min_lng' | 'geofence_max_lng'>): Box {
  return { minLat: s.geofence_min_lat, maxLat: s.geofence_max_lat, minLng: s.geofence_min_lng, maxLng: s.geofence_max_lng }
}

/** Inclusive, like lights.in_geofence(). */
export function inBox(p: LatLng, b: Box): boolean {
  return p.lat >= b.minLat && p.lat <= b.maxLat && p.lng >= b.minLng && p.lng <= b.maxLng
}

export function boxCenter(b: Box): LatLng {
  return { lat: (b.minLat + b.maxLat) / 2, lng: (b.minLng + b.maxLng) / 2 }
}

/** The box grown by `meters` on every side (the mini-map's pan limit, so the pin can
 *  still reach the geofence edge). */
export function padBox(b: Box, meters: number): Box {
  const dLat = meters / 111_320
  const dLng = meters / (111_320 * Math.cos((((b.minLat + b.maxLat) / 2) * Math.PI) / 180))
  return { minLat: b.minLat - dLat, maxLat: b.maxLat + dLat, minLng: b.minLng - dLng, maxLng: b.maxLng + dLng }
}

/** Six decimals (~10 cm): plenty for a house, and stable through JSON round trips. */
export function roundPoint(p: LatLng): LatLng {
  const r = (v: number) => Math.round(v * 1e6) / 1e6
  return { lat: r(p.lat), lng: r(p.lng) }
}

/** Great-circle distance in meters (haversine). */
export function distanceMeters(a: LatLng, b: LatLng): number {
  const rad = Math.PI / 180
  const dLat = (b.lat - a.lat) * rad
  const dLng = (b.lng - a.lng) * rad
  const h = Math.sin(dLat / 2) ** 2 + Math.cos(a.lat * rad) * Math.cos(b.lat * rad) * Math.sin(dLng / 2) ** 2
  return 2 * 6_371_000 * Math.asin(Math.sqrt(h))
}

/** Google Maps at a point (the admin's "check the pin" link). */
export function mapsSearchUrl(p: LatLng): string {
  return `https://www.google.com/maps/search/?api=1&query=${p.lat.toFixed(6)},${p.lng.toFixed(6)}`
}
````

````ts
// Pure rules for the entry form, mirroring the database (M3 D6) so mistakes show up
// before anything is uploaded. The database stays the authority.

export const ADDRESS_MIN = 5
export const ADDRESS_MAX = 200
export const TITLE_MAX = 60

/** Collapse whitespace and trim. */
export const tidy = (text: string) => text.replace(/\s+/g, ' ').trim()

export type AddressProblem = 'empty' | 'short' | 'long' | 'number'

/** Why an address can't be submitted, or null if it can. A house number is required:
 *  the geocoder needs it and every house in the Hills has one. */
export function addressProblem(raw: string): AddressProblem | null {
  const a = tidy(raw)
  if (!a) return 'empty'
  if (a.length < ADDRESS_MIN) return 'short'
  if (a.length > ADDRESS_MAX) return 'long'
  if (!/^\d+[a-z]?\s/i.test(a)) return 'number'
  return null
}

/** The optional display name: tidied, null when empty. */
export function titleValue(raw: string): string | null {
  const t = tidy(raw)
  return t ? t : null
}

export const titleTooLong = (raw: string) => tidy(raw).length > TITLE_MAX

/** A new, unique object name in the owner's folder. The database accepts only
 *  `<uid>/<[A-Za-z0-9][A-Za-z0-9._-]*>` (M3 R3) and nobody can overwrite an object, so every
 *  upload gets a fresh name (which also keeps the CDN from serving a stale photo). */
export function photoPath(uid: string, now: number, random: string): string {
  return `${uid}/${now}-${random.replace(/[^A-Za-z0-9]/g, '').slice(0, 12)}.jpg`
}

/** The object name inside the `houses` bucket for one of its public URLs, or null. */
export function objectNameFromUrl(url: string): string | null {
  const marker = '/storage/v1/object/public/houses/'
  const i = url.indexOf(marker)
  if (i < 0) return null
  const name = url.slice(i + marker.length)
  return /^[0-9a-f-]{36}\/[A-Za-z0-9][A-Za-z0-9._-]*$/.test(name) ? name : null
}

/** What went wrong with a write, in terms the UI has copy for (M3 D26). */
export type EntryErrorKind =
  | 'already-entered' // 23505 on houses_submitted_by_key: one house per phone
  | 'duplicate-address' // 23505 on houses_address_key_key
  | 'rejected' // 42501: window closed, outside the geofence, or the photo rule
  | 'photo-too-large' // storage 413
  | 'network' // the request never got an answer
  | 'unknown'

interface ErrorLike {
  code?: unknown
  message?: unknown
  statusCode?: unknown
  status?: unknown
  name?: unknown
}

export function entryErrorKind(error: unknown): EntryErrorKind {
  const e = (error ?? {}) as ErrorLike
  const message = typeof e.message === 'string' ? e.message : ''
  const code = typeof e.code === 'string' ? e.code : ''
  if (code === '23505' && message.includes('houses_submitted_by_key')) return 'already-entered'
  if (code === '23505' && message.includes('houses_address_key_key')) return 'duplicate-address'
  if (code === '42501' || /row-level security/i.test(message)) return 'rejected'
  const status = Number(e.statusCode ?? e.status)
  if (status === 413 || /exceeded the maximum allowed size/i.test(message)) return 'photo-too-large'
  if (error instanceof TypeError || /failed to fetch|network|load failed/i.test(message)) return 'network'
  return 'unknown'
}
````

````ts
import type { AnySupabaseClient } from '@wohills/core'
import type { LatLng } from './geo'

export type LocateError = 'denied' | 'unavailable' | 'timeout' | 'unsupported'

/** One GPS fix. Rejects with a LocateError. High accuracy because we want the house,
 *  not the block; a fix up to a minute old is fine (they're standing in the driveway). */
export function currentPosition(timeoutMs = 15_000): Promise<LatLng> {
  return new Promise((resolve, reject) => {
    if (!('geolocation' in navigator)) {
      reject('unsupported' satisfies LocateError)
      return
    }
    navigator.geolocation.getCurrentPosition(
      (pos) => resolve({ lat: pos.coords.latitude, lng: pos.coords.longitude }),
      (err) => reject((err.code === err.PERMISSION_DENIED ? 'denied' : err.code === err.TIMEOUT ? 'timeout' : 'unavailable') satisfies LocateError),
      { enableHighAccuracy: true, timeout: timeoutMs, maximumAge: 60_000 },
    )
  })
}

export type GeocodeResult =
  | { kind: 'match'; point: LatLng; matchedAddress: string; inGeofence: boolean }
  | { kind: 'no-match' }
  | { kind: 'error'; reason: 'busy' | 'unavailable' | 'signed-out' | 'invalid' | 'network' }

/** Pure: the geocode function's answer (M3 D17) as a result the form can show. */
export function geocodeOutcome(status: number | 'network', body: unknown): GeocodeResult {
  if (status === 'network') return { kind: 'error', reason: 'network' }
  if (status === 200) {
    const b = body as { match?: { lat?: unknown; lng?: unknown; matchedAddress?: unknown }; inGeofence?: unknown } | null
    const m = b?.match
    if (typeof m?.lat === 'number' && typeof m.lng === 'number')
      return {
        kind: 'match',
        point: { lat: m.lat, lng: m.lng },
        matchedAddress: typeof m.matchedAddress === 'string' ? m.matchedAddress : '',
        inGeofence: b?.inGeofence === true,
      }
    return { kind: 'error', reason: 'unavailable' }
  }
  if (status === 404) return { kind: 'no-match' }
  if (status === 400) return { kind: 'error', reason: 'invalid' }
  if (status === 401) return { kind: 'error', reason: 'signed-out' }
  if (status === 429) return { kind: 'error', reason: 'busy' }
  return { kind: 'error', reason: 'unavailable' }
}

/** Typed address → point, through the `geocode` Edge Function (US Census, server-side
 *  geofence check). Needs a signed-in phone account; supabase-js sends its token. */
export async function geocodeAddress(client: AnySupabaseClient, address: string): Promise<GeocodeResult> {
  const { data, error } = await client.functions.invoke('geocode', { body: { address } })
  if (!error) return geocodeOutcome(200, data)
  const response = (error as { context?: unknown }).context
  if (response instanceof Response) return geocodeOutcome(response.status, await response.json().catch(() => null))
  return geocodeOutcome('network', null)
}
````

### A.12: `apps/lights/src/lib/submit.ts`

````ts
import { uploadPublicImage, type AnySupabaseClient } from '@wohills/core'
import type { EntryRow } from '../types'
import { photoPath } from './entry'
import { roundPoint, type LatLng } from './geo'

export const ENTRY_COLUMNS = 'id, entry_number, title, address, lat, lng, photo_url, submitted_by, status, created_at, updated_at'

const BUCKET = 'houses'

/** A photo that went through prepareImageForUpload() (resized JPEG, EXIF stripped). */
export interface PreparedPhoto {
  blob: Blob
  contentType: string
}

/** What the form produces. `photo` is null when an edit keeps the current photo. */
export interface EntryDraft {
  photo: PreparedPhoto | null
  address: string
  point: LatLng
  title: string | null
}

/** The owner's edit was refused because owner edits are locked (voting opened, or entries
 *  closed): RLS filters the row, so PostgREST answers with zero rows and no error (M3 D9). */
export class EntryLockedError extends Error {
  constructor() {
    super('entry is locked')
    this.name = 'EntryLockedError'
  }
}

export type Stage = 'upload' | 'save'

interface Options {
  onStage?: (stage: Stage) => void
  now?: () => number
  random?: () => string
}

/** Columns an edit would change (an empty object means nothing to save). Pure. */
export function changedFields(house: EntryRow, draft: EntryDraft): Partial<Pick<EntryRow, 'address' | 'lat' | 'lng' | 'title'>> {
  const p = roundPoint(draft.point)
  const out: Partial<Pick<EntryRow, 'address' | 'lat' | 'lng' | 'title'>> = {}
  if (draft.address !== house.address) out.address = draft.address
  if (p.lat !== house.lat || p.lng !== house.lng) {
    out.lat = p.lat
    out.lng = p.lng
  }
  if (draft.title !== house.title) out.title = draft.title
  return out
}

async function upload(client: AnySupabaseClient, uid: string, photo: PreparedPhoto, o: Options) {
  const path = photoPath(uid, (o.now ?? Date.now)(), (o.random ?? (() => crypto.randomUUID()))())
  const url = await uploadPublicImage(client, { bucket: BUCKET, path, ...photo })
  return { path, url }
}

/** Best effort: storage refuses (silently, zero rows) whatever the owner may not delete. */
async function removeQuietly(client: AnySupabaseClient, paths: string[]) {
  if (paths.length === 0) return
  try {
    await client.storage.from(BUCKET).remove(paths)
  } catch {
    // An orphan in the owner's own folder is harmless (unlisted, unguessable name).
  }
}

/** Remove every object in the owner's folder except the photo their entry shows now,
 *  which the storage policy protects anyway (M3 R2). Clears photos left by a replaced
 *  photo, a withdrawn entry or an attempt that died between upload and save. Works only
 *  while owner edits are allowed (before voting opens); after that the policy keeps them. */
export async function sweepOrphans(client: AnySupabaseClient, uid: string, keep: string | null) {
  try {
    const { data } = await client.storage.from(BUCKET).list(uid, { limit: 100 })
    const paths = (data ?? []).map((f) => `${uid}/${f.name}`).filter((p) => p !== keep)
    await removeQuietly(client, paths)
  } catch {
    // Best effort.
  }
}

/** New entry. Order matters (M3 "Notes"): the photo must already exist in the owner's
 *  folder when the row is inserted, so upload first, then insert; if the insert fails,
 *  delete the upload again. Throws the Supabase error (see entryErrorKind). */
export async function submitEntry(client: AnySupabaseClient, uid: string, draft: EntryDraft, o: Options = {}): Promise<EntryRow> {
  if (!draft.photo) throw new Error('a new entry needs a photo')
  o.onStage?.('upload')
  const { path, url } = await upload(client, uid, draft.photo, o)
  o.onStage?.('save')
  const p = roundPoint(draft.point)
  const { data, error } = await client
    .from('houses')
    .insert({ address: draft.address, lat: p.lat, lng: p.lng, photo_url: url, title: draft.title })
    .select(ENTRY_COLUMNS)
    .single<EntryRow>()
  if (error) {
    await removeQuietly(client, [path])
    throw error
  }
  await sweepOrphans(client, uid, path)
  return data
}

/** The owner's edit. A new photo is uploaded first, then the row points at it, then the
 *  old photo is removed: storage won't let an owner delete the photo their entry still
 *  shows (M3 R2), and the row can't point at a photo that isn't uploaded yet (M3 R3).
 *  Any change sends an approved entry back to pending (the database trigger does that). */
export async function updateEntry(client: AnySupabaseClient, house: EntryRow, draft: EntryDraft, o: Options = {}): Promise<EntryRow> {
  const uid = house.submitted_by
  const patch: Record<string, unknown> = changedFields(house, draft)
  let uploaded: string | null = null
  if (draft.photo) {
    o.onStage?.('upload')
    const { path, url } = await upload(client, uid, draft.photo, o)
    uploaded = path
    patch.photo_url = url
  }
  if (Object.keys(patch).length === 0) return house
  o.onStage?.('save')
  const { data, error } = await client.from('houses').update(patch).eq('id', house.id).select(ENTRY_COLUMNS)
  if (error || !data?.length) {
    if (uploaded) await removeQuietly(client, [uploaded])
    throw error ?? new EntryLockedError()
  }
  if (uploaded) await sweepOrphans(client, uid, uploaded)
  return data[0] as EntryRow
}

/** Withdraw: delete the row first (that cascades nothing before voting opens: there are no
 *  votes yet), then its photos, which storage only allows once no entry shows them. */
export async function withdrawEntry(client: AnySupabaseClient, house: EntryRow): Promise<void> {
````

````ts
/** Whether the signed-in phone is the admin: undefined while asking (core's useIsAdmin

export function useAdminCheck(session: Session | null): boolean | undefined {

  const [answer, setAnswer] = useState<{ uid: string; admin: boolean } | null>(null)
  useEffect(() => {
    if (!uid) return

      .schema('public')
      .rpc('is_admin')
      .then(({ data }) => {
        if (!cancelled) setAnswer({ uid, admin: data === true })
      })
    return () => {
      cancelled = true
    }
  }, [client, uid])
  if (!uid) return false
  return answer?.uid === uid ? answer.admin : undefined
}

 *  Status changes are optimistic: the row moves to its new list at once, and goes back
 *  (with the error rethrown) if the database refuses. */
export function useAdminEntries(enabled: boolean) {
  const client = useSupabase()
  const { data, loading, refresh } = usePolling(
    async () => {
      const { data, error } = await client.from('houses').select(ENTRY_COLUMNS).order('created_at')
      if (error) throw error
      return data as EntryRow[]
    },
    { enabled },
  )
  const [optimistic, setOptimistic] = useState<Record<string, HouseStatus>>({})
  const rows = data?.map((r) => (optimistic[r.id] ? { ...r, status: optimistic[r.id] } : r))

    async (id: string, status: HouseStatus) => {
      setOptimistic((o) => ({ ...o, [id]: status }))

        await refresh()

          const next = { ...o }
          delete next[id]
          return next
        })
      }
    },
    [client, refresh],
  )
  return { rows, loading, refresh, setStatus }
}

/** The admin's edit of any entry, at any time (M3 D13). A replacement photo goes into the
 *  owner's folder as `admin-…` (the admin is exempt from the folder rule), then the row
 *  points at it, then the old photo is deleted. Status is left alone: the admin decides. */
export async function adminEditEntry(
  client: AnySupabaseClient,
  row: EntryRow,
  edit: AdminEdit,
  { now = Date.now, random = () => crypto.randomUUID() }: { now?: () => number; random?: () => string } = {},
): Promise<void> {

  const patch: Record<string, unknown> = {}

  if (edit.address !== row.address) patch.address = edit.address
  if (p.lat !== row.lat || p.lng !== row.lng) Object.assign(patch, p)
  let uploaded: string | null = null
  if (edit.photo) {
    uploaded = photoPath(row.submitted_by, now(), random()).replace(/\/(?=[^/]+$)/, '/admin-')
    patch.photo_url = await uploadPublicImage(client, { bucket: 'houses', path: uploaded, ...edit.photo })
  }
  if (Object.keys(patch).length === 0) return
  const { data, error } = await client.from('houses').update(patch).eq('id', row.id).select('id')
  if (error || !data?.length) {
    if (uploaded) await client.storage.from('houses').remove([uploaded])
    throw error ?? new Error('not allowed')
  }
  const old = objectNameFromUrl(row.photo_url)
  if (uploaded && old) await client.storage.from('houses').remove([old])
}

  if (paths.length) await client.storage.from('houses').remove(paths)
}
````

````tsx
/** The admin's edit: name, address, pin (numbers; "Check the pin" opens Google Maps) and a
 *  replacement photo. Not geofenced and no re-approval: the admin is trusted (M3 D13). */
export function AdminEditDialog({ row, onSave, onClose }: { row: EntryRow; onSave: (edit: AdminEdit) => Promise<void>; onClose: () => void }) {
  const id = useId()
  const [title, setTitle] = useState(row.title ?? '')
  const [address, setAddress] = useState(row.address)
  const [lat, setLat] = useState(String(row.lat))
  const [lng, setLng] = useState(String(row.lng))
  const [photo, setPhoto] = useState<(PreparedPhoto & { previewUrl: string }) | null>(null)
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    if (!photo) return
    return () => URL.revokeObjectURL(photo.previewUrl)
  }, [photo])

  async function pick(file: File) {
    setError(null)
    try {
      const prepared = await prepareImageForUpload(file)
      setPhoto({ ...prepared, previewUrl: URL.createObjectURL(prepared.blob) })
    } catch {
      setError(ENTER.photoErrors.other)
    }
  }

  async function submit(e: FormEvent) {
    e.preventDefault()
    const a = tidy(address)
    const point = { lat: Number(lat), lng: Number(lng) }
    if (a.length < ADDRESS_MIN || a.length > ADDRESS_MAX) return setError(ADMIN.addressInvalid)
    if (tidy(title).length > TITLE_MAX) return setError(ENTER.titleTooLong(TITLE_MAX))
    if (!Number.isFinite(point.lat) || !Number.isFinite(point.lng) || Math.abs(point.lat) > 90 || Math.abs(point.lng) > 180)
      return setError(ADMIN.pinInvalid)
    setBusy(true)
    setError(null)
    try {
      await onSave({ title: titleValue(title), address: a, point, photo })
    } catch {
      setError(ADMIN.toasts.failed)
      setBusy(false)
    }
  }

  return (
    <Dialog label={ADMIN.editTitle(row.entry_number)} onClose={() => !busy && onClose()}>
      <h2 data-slot="dialog-title">{ADMIN.editTitle(row.entry_number)}</h2>
      <form onSubmit={submit} className="mt-6 space-y-5">
        <div>
          <label htmlFor={`${id}-title`} data-slot="field-label">
            {ENTER.titleLabel}
          </label>
          <input id={`${id}-title`} data-slot="field" value={title} onChange={(e) => setTitle(e.target.value)} />
        </div>
        <div>
          <label htmlFor={`${id}-address`} data-slot="field-label">
            {ENTER.addressLabel}
          </label>
          <input id={`${id}-address`} data-slot="field" value={address} onChange={(e) => setAddress(e.target.value)} />
        </div>
        <div className="grid grid-cols-2 gap-3">
          <div>
            <label htmlFor={`${id}-lat`} data-slot="field-label">
              {ADMIN.latLabel}
            </label>
            <input id={`${id}-lat`} data-slot="field" inputMode="decimal" value={lat} onChange={(e) => setLat(e.target.value)} />
          </div>
          <div>
            <label htmlFor={`${id}-lng`} data-slot="field-label">
              {ADMIN.lngLabel}
            </label>
            <input id={`${id}-lng`} data-slot="field" inputMode="decimal" value={lng} onChange={(e) => setLng(e.target.value)} />
          </div>
        </div>
        <div className="flex flex-wrap items-center gap-4">
          <img src={photo?.previewUrl ?? row.photo_url} alt="" className="h-20 w-28 flex-none bg-sunken object-cover" />
          <div className="space-y-2">
            <PhotoPicker onFile={pick} disabled={busy} className="btn-ghost">
              {ADMIN.replacePhoto}
            </PhotoPicker>
            {photo && <p className="text-sm text-fg/75">{ADMIN.photoReady}</p>}
          </div>
        </div>
        {error && (
          <Note tone="error" live="alert">
            {error}
          </Note>
        )}
        <div className="flex flex-wrap gap-3 pt-2">
          <button type="submit" className="btn-bulb" disabled={busy}>
            {ADMIN.save}
          </button>
          <button type="button" className="btn-ghost" disabled={busy} onClick={onClose}>
            {ADMIN.cancel}
          </button>
        </div>
      </form>
    </Dialog>
  )
}
````

````ts
/** `animation:` shorthands and `animation-duration`s in effects.css, as [name, seconds, steps]. */
function animations(css: string) {
  const out: { decl: string; seconds: number; steps: number | null }[] = []
  for (const m of css.matchAll(/animation:\s*([^;]+);/g)) {
    const decl = m[1].trim()
    if (decl.startsWith('none')) continue
    const seconds = Number(decl.match(/(\d*\.?\d+)s\b/)?.[1])
    const keyframes = decl.split(/\s+/)[0]
    // A step-end chase changes once per keyframe; count the keyframe stops.
    const block = css.match(new RegExp(`@keyframes ${keyframes}\\s*\\{([\\s\\S]*?)\\n\\}`))?.[1] ?? ''
    const steps = /step-end|steps\(/.test(decl) ? (block.match(/\d+%/g)?.length ?? 1) - 1 : null
    out.push({ decl, seconds, steps })
  }
  return out
}
````

````ts
/** A signed-in client for a test phone. GoTrue allows one code per phone every few
 *  seconds, so this waits out the cooldown instead of failing. */
export async function signInApi(phone: string): Promise<{ client: Client; session: Session }> {
  const { url, anonKey } = env()
  const client = createClient(url, anonKey, opts)
  let sent = await client.auth.signInWithOtp({ phone })
  for (let i = 0; i < 4 && sent.error?.code === 'over_sms_send_rate_limit'; i++) {
    await new Promise((resolve) => setTimeout(resolve, 5000))
    sent = await client.auth.signInWithOtp({ phone })
  }
  if (sent.error) throw sent.error
  const { data, error } = await client.auth.verifyOtp({ phone, token: '123456', type: 'sms' })
  if (error || !data.session) throw error ?? new Error('no session')
  return { client, session: data.session }
}

 *  reach the Census geocoder from this sandbox; M3 D19). `geocode` maps a lower-case

export async function stubNetwork(page: Page, geocode: Record<string, { lat: number; lng: number; inGeofence: boolean }> = {}) {

    const address = String((route.request().postDataJSON() as { address?: string })?.address ?? '').toLowerCase()

/** Drag the pin map by (dx, dy) pixels with one finger (the tests run as a touch phone,
 *  where MapLibre pans on touch events; Playwright's mouse would not move it). */

  const map = page.locator('[data-slot="mini-map"]')
  await map.scrollIntoViewIfNeeded()
  const box = (await map.boundingBox())!
  const x = box.x + box.width / 2
  const y = box.y + box.height / 2
  const cdp = await page.context().newCDPSession(page)
  await cdp.send('Input.dispatchTouchEvent', { type: 'touchStart', touchPoints: [{ x, y }] })
  for (let i = 1; i <= 10; i++)
    await cdp.send('Input.dispatchTouchEvent', { type: 'touchMove', touchPoints: [{ x: x + (dx * i) / 10, y: y + (dy * i) / 10 }] })
  await cdp.send('Input.dispatchTouchEvent', { type: 'touchEnd', touchPoints: [] })
  await cdp.detach()
}
````

````ts
async function checkGuards(page: Page, screen: string) {
````

## Appendix F: extraction script (save as `$S/m5/extract.py`; never commit)

### F.1 extract.py

````python
#!/usr/bin/env python3
"""Write the M5 appendices into the repo.

Usage (from the repo root): python3 extract.py docs/milestones/M5.md

Every "### X.n: `path`" heading in Appendices A-E is followed by one fenced block (four
backticks), sometimes after a one-line note. Plain headings write the file, "(append)" appends the block once, "(diff)"
applies it with `git apply` (skipped if already applied). Safe to run twice.
"""
import os
import re
import subprocess
import sys
import tempfile

text = open(sys.argv[1], encoding='utf-8').read()
pattern = re.compile(
    r'^### ([A-E])\.(\d+): `([^`]+)`( \((append|diff)\))?\n\n(?:[^`\n][^\n]*\n\n)?````[a-z]*\n(.*?)\n````$', re.S | re.M
)
count = 0
for m in pattern.finditer(text):
    path, mode, body = m.group(3), m.group(5), m.group(6) + '\n'
    if mode == 'diff':
        with tempfile.NamedTemporaryFile('w', suffix='.diff', delete=False) as f:
            f.write(body)
        if subprocess.run(['git', 'apply', '--check', f.name], capture_output=True).returncode == 0:
            subprocess.run(['git', 'apply', f.name], check=True)
            print('patched ', path)
        elif subprocess.run(['git', 'apply', '-R', '--check', f.name], capture_output=True).returncode == 0:
            print('ok      ', path, '(already patched)')
        else:
            sys.exit(f'cannot apply the diff for {path}')
        os.unlink(f.name)
    elif mode == 'append':
        current = open(path, encoding='utf-8').read()
        if body.strip() in current:
            print('ok      ', path, '(already appended)')
        else:
            with open(path, 'a', encoding='utf-8') as f:
                f.write(body)
            print('appended', path)
    else:
        os.makedirs(os.path.dirname(path) or '.', exist_ok=True)
        with open(path, 'w', encoding='utf-8') as f:
            f.write(body)
        print('wrote   ', path)
    count += 1
print(count, 'appendix files')
````
