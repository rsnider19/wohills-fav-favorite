## Goal

Neighbors vote for the brightest house, and the winners light up:

- **Voting** on every lineup card and in the map's house sheet, through the shared hooks
  (`useMyVote`, `useResults`) and the shared `VoteButton`: one vote per phone, changeable
  until the close, never for your own house. A signed-out tap signs you in and then casts that
  vote (the parade's pattern). The card you voted for goes fully lit.
- **Every window state** says what it is: "Voting opens Tue, Dec 8" before, the countdown
  while live, "Voting paused" if the admin pauses, "Voting closed" and "Votes are in" after,
  all flipping at the boundary without a reload. Late-approved houses join mid-vote on the
  30 s poll.
- **The reveal**: when the admin sets `RESULTS_VISIBLE=true`, open pages pick it up on their
  next poll and the top 3 appear on a podium in the hero (rank, photo, number, name, address,
  vote count, directions), with the signature **power surge**: the page dims, then 3rd, 2nd
  and 1st come on one at a time with a brownout flicker. WCAG 2.3.1-safe (measured), static
  under reduced motion and the pause switch, once per browser with Skip and Replay.
- **The admin's `?preview=results`** shows the podium and the surge before the reveal with a
  sample podium, never a real count.
- **Done when** (plan §6 M7): the full window lifecycle works locally (pre-vote → live →
  closed → revealed), proven by the e2e tests and the preview toggle. No SQL change.

control is ≥ 44 px; new contrast pairs are pinned.

### A.1: `apps/lights/src/lib/ballot.ts`

````ts
import type { Phase } from './phase'
import type { House } from '../types'

/** What one house's vote control shows this visitor (M7). Display only: RLS decides
 *  (M3 D10). Each state maps onto the shared <VoteButton> in components/Ballot.tsx. */
export type BallotState =
  /** Voting is open and nobody is signed in: "Sign in to vote". */
  | 'sign-in'
  /** Open, signed in, no counted pick yet. */
  | 'vote'
  /** Open, signed in, the pick is another house: "Switch my vote here". */
  | 'change'
  /** This house is the visitor's pick (open, closed or paused). */
  | 'pick'
  /** The visitor's own house: self-votes are blocked (plan §7 #4). */
  | 'own'
  /** Before voting opens. */
  | 'not-yet'
  /** Inside the window with the kill switch off. */
  | 'paused'
  /** After the close, waiting for the reveal. */
  | 'closed'
  /** After the reveal: no vote control at all. */
  | 'none'

export interface BallotInput {
  house: Pick<House, 'id' | 'submitted_by'>
  /** The signed-in user's id, or null. */
  uid: string | null
  /** The visitor's counted pick: their vote, but only while that house is still approved
   *  (see `countedPick`). */
  pick: string | null
  phase: Phase
  votingOpen: boolean
  votingPaused: boolean
}

/** Pure: the vote control for one house. Order matters: the reveal hides every control;
 *  your own house never offers a vote; a pick stays lit after the close. */
export function ballotState({ house, uid, pick, phase, votingOpen, votingPaused }: BallotInput): BallotState {
  if (phase === 'results') return 'none'
  if (uid !== null && house.submitted_by === uid) return 'own'
  if (pick === house.id) return 'pick'
  if (votingOpen) return uid === null ? 'sign-in' : pick === null ? 'vote' : 'change'
  if (votingPaused) return 'paused'
  return phase === 'before' || phase === 'submissions' ? 'not-yet' : 'closed'
}

/** Pure: the vote that counts. A vote for a house the admin hid is kept but not counted
 *  (M3 D11); the voter sees no pick and may vote again. `houses` undefined = not loaded
 *  yet, so nothing is known and the stored vote is trusted. */
export function countedPick(myVote: string | null, houses: Pick<House, 'id'>[] | undefined): string | null {
  if (myVote === null || houses === undefined) return myVote
  return houses.some((h) => h.id === myVote) ? myVote : null
}

/** Pure: the visitor voted for a house that is no longer on the board. */
export function pickWasHidden(myVote: string | null, houses: Pick<House, 'id'>[] | undefined): boolean {
  return myVote !== null && houses !== undefined && countedPick(myVote, houses) === null
}

/** Why a vote didn't go in, for the toast. RLS answers every refusal with the same 42501
 *  (M3 D26), so the app asks the database what changed (`facts`) before choosing the copy. */
export type VoteProblem = 'closed' | 'gone' | 'own' | 'network' | 'unknown'

export interface VoteFacts {
  /** lights.voting_is_open() right after the refusal (null if that check failed too). */
  votingOpen: boolean | null
  /** The house is still approved (null if unknown). */
  approved: boolean | null
  /** The house belongs to the voter. */
  own: boolean
}

interface ErrorLike {
  code?: unknown
  message?: unknown
}

/** Pure: a network failure (supabase-js reports a failed fetch as code '' with the
 *  TypeError's message), as opposed to an answer from the database. */
export function isNetworkError(error: unknown): boolean {
  if (error instanceof TypeError) return true
  const e = (error ?? {}) as ErrorLike
  const message = typeof e.message === 'string' ? e.message : ''
  return /failed to fetch|network|load failed/i.test(message)
}

/** Pure: the vote problem, from the error and what the database says now. */
export function voteProblem(error: unknown, facts: VoteFacts | null): VoteProblem {
  if (isNetworkError(error)) return 'network'
  const e = (error ?? {}) as ErrorLike
  const rejected = e.code === '42501' || (typeof e.message === 'string' && /row-level security/i.test(e.message))
  if (!rejected || !facts) return 'unknown'
  if (facts.own) return 'own'
  if (facts.votingOpen === false) return 'closed'
  if (facts.approved === false) return 'gone'
  return 'unknown'
}
````

### A.3: `apps/lights/src/lib/reveal.ts`

````ts
import type { House } from '../types'

/** One row of lights.get_results() (M3 D12): nothing before the reveal, then at most the
 *  top 3 approved houses, ranked by votes with the earlier entry first on a tie. */
export interface LightsResult {
  house_id: string
  entry_number: number
  title: string | null
  address: string
  photo_url: string
  vote_count: number
  rank: number
}

/** A house on the podium. `votes` is null on the admin's sample podium (no real counts). */
export interface PodiumRow {
  id: string
  rank: number
  entryNumber: number
  title: string | null
  address: string
  photoUrl: string
  votes: number | null
  /** Another house on the podium has the same count (the SQL put the earlier entry first). */
  tied: boolean
  /** The approved house, for directions and "Your house" (get_results has no lat/lng). */
  house: House | null
}

/** Pure: the podium from get_results(). Houses with no votes aren't winners, so they are
 *  left off (a podium of 0-vote houses would be ranked by entry number alone). */
export function podiumRows(results: LightsResult[], houses: House[] | undefined): PodiumRow[] {
  const rows = results
    .filter((r) => Number(r.vote_count) > 0)
    .sort((a, b) => Number(a.rank) - Number(b.rank))
    .slice(0, 3)
  return rows.map((r) => ({
    id: r.house_id,
    rank: Number(r.rank),
    entryNumber: r.entry_number,
    title: r.title,
    address: r.address,
    photoUrl: r.photo_url,
    votes: Number(r.vote_count),
    tied: rows.some((o) => o !== r && Number(o.vote_count) === Number(r.vote_count)),
    house: houses?.find((h) => h.id === r.house_id) ?? null,
  }))
}

/** Pure: the admin's preview podium before the reveal (?preview=results). get_results()
 *  returns nothing until results_visible (even for the admin, M3 lights_30), so this uses
 *  the first three approved houses by entry number, with no counts: nothing real leaks. */
export function samplePodium(houses: House[]): PodiumRow[] {
  return [...houses]
    .sort((a, b) => a.entry_number - b.entry_number)
    .slice(0, 3)
    .map((h, i) => ({
      id: h.id,
      rank: i + 1,
      entryNumber: h.entry_number,
      title: h.title,
      address: h.address,
      photoUrl: h.photo_url,
      votes: null,
      tied: false,
      house: h,
    }))
}

/** Pure: identifies this podium. The reveal plays once per browser per podium; if the
 *  admin changes it after the reveal (hides a winner), it plays again. */
export const revealKey = (rows: PodiumRow[]) => rows.map((r) => `${r.id}:${r.votes ?? '?'}`).join(',')

const SEEN = 'lights-reveal-seen'

/** Whether this browser already watched the reveal of this podium. */
export function seenReveal(key: string): boolean {
  try {
    return localStorage.getItem(SEEN) === key
  } catch {
    return false // storage blocked (private mode): play it, then show the lit podium
  }
}

export function markRevealSeen(key: string) {
  try {
    localStorage.setItem(SEEN, key)
  } catch {
    // Private mode: it plays again next visit, which is harmless.
  }
}

/** The "power surge", in ms (effects.css section 15 holds the same numbers). The page
 *  dims, then No. 3, No. 2 and No. 1 come on one after another with a brownout flicker,
 *  then the power comes back. `surge` is when each place starts. */
export const REVEAL = {
  total: 7000,
  surge: { 3: 900, 2: 2500, 1: 4100 } as Record<number, number>,
}
````

````tsx
import { useEffect, useId, useMemo, useRef, useState, useSyncExternalStore, type CSSProperties } from 'react'
import { GALLERY, PODIUM, VOTE } from '../copy'
import { markRevealSeen, REVEAL, revealKey, seenReveal, type PodiumRow } from '../lib/reveal'
import { NavigateButton } from './NavigateButton'

interface Props {
  rows: PodiumRow[]
  /** get_results() hasn't answered yet. */
  loading: boolean
  /** The admin's ?preview=results before the real reveal: sample houses, no counts. */
  sample: boolean
  /** The header's twinkle switch. Off = no reveal animation, the lit podium at once. */
  motionOn: boolean
  /** For the "Your pick" / "Your house" chips. */
  uid: string | null
  pick: string | null
}

const REDUCED = '(prefers-reduced-motion: reduce)'
function subscribeReduced(onChange: () => void) {
  const mq = matchMedia(REDUCED)
  mq.addEventListener('change', onChange)
  return () => mq.removeEventListener('change', onChange)
}
const reducedNow = () => matchMedia(REDUCED).matches

/** The reveal's state for one podium. `armed`: waiting (dark) until the podium is on
 *  screen; `play`: the power surge is running; `done`: lit. */
type Mode = 'wait' | 'armed' | 'play' | 'done'

/** The top 3 after the reveal, inside the hero (M4 slot), with the "power surge": the
 *  page dims, No. 3, No. 2 and No. 1 come on with a brownout flicker, the power returns.
 *  The animation is CSS only (effects.css section 15); nothing here waits for
 *  `animationend`, so reduced motion, the pause switch, Skip or scrolling away simply
 *  show the final, lit state (every base style is the lit one, M4 D12). */
export function Podium({ rows, loading, sample, motionOn, uid, pick }: Props) {
  const headingId = useId()
  const section = useRef<HTMLElement>(null)
  const heading = useRef<HTMLHeadingElement>(null)
  const reduced = useSyncExternalStore(subscribeReduced, reducedNow, () => false)
  const key = revealKey(rows)
  // Read once per podium: a later markRevealSeen() must not end a reveal that is playing.
  const seen = useMemo(() => !sample && seenReveal(key), [sample, key])
  const [run, setRun] = useState<{ key: string; mode: 'play' | 'done' } | null>(null)

  // No IntersectionObserver (very old browsers): never arm, or the podium would stay dark.
  const animate = motionOn && !reduced && typeof IntersectionObserver !== 'undefined'
  let mode: Mode
  if (loading || rows.length === 0) mode = 'wait'
  else if (!animate) mode = 'done'
  else if (run?.key === key) mode = run.mode
  else mode = seen ? 'done' : 'armed'

  // Armed: start once the podium's top is in the upper 60% of the screen (on a phone it
  // sits below the hero, so the surge waits for the visitor instead of playing off screen).
  // Measured against the viewport, not the podium's own height, so a tall podium on a
  // short or landscape screen still starts.
  useEffect(() => {
    const el = section.current
    if (mode !== 'armed' || !el) return
    const io = new IntersectionObserver(
      ([entry]) => {
        if (!entry.isIntersecting) return
        if (!sample) markRevealSeen(key)
        setRun({ key, mode: 'play' })
      },
      { rootMargin: '0px 0px -40% 0px' },
    )
    io.observe(el)
    return () => io.disconnect()
  }, [mode, key, sample])

  // Playing: done after the last surge, or as soon as the podium leaves the screen (the
  // dimmed page must never linger while the visitor reads the lineup).
  useEffect(() => {
    const el = section.current
    if (mode !== 'play' || !el) return
    const finish = () => setRun({ key, mode: 'done' })
    const timer = setTimeout(finish, REVEAL.total)
    const io = new IntersectionObserver(([entry]) => {
      if (!entry.isIntersecting) finish()
    })
    io.observe(el)
    return () => {
      clearTimeout(timer)
      io.disconnect()
    }
  }, [mode, key])

  const toggle = () => {
    if (mode === 'play') {
      setRun({ key, mode: 'done' })
      heading.current?.focus({ preventScroll: true })
    } else setRun({ key, mode: 'play' })
  }

  return (
    <section
      ref={section}
      id="winners"
      data-slot="podium"
      data-reveal={mode === 'play' || mode === 'armed' ? mode : undefined}
      data-sample={sample || undefined}
      aria-labelledby={headingId}
      aria-busy={mode === 'wait' && loading ? true : undefined}
    >
      {mode === 'play' && <div data-slot="blackout" aria-hidden="true" />}
      <div className="mx-auto w-full max-w-6xl px-4 pb-14 sm:px-6 lg:pb-20">
        <div className="mb-8 flex flex-wrap items-end justify-between gap-4">
          <div>
            <p className="neon-sm mb-2 font-display text-xs tracking-[0.2em] text-primary uppercase sm:text-sm">
              {sample ? PODIUM.sampleEyebrow : PODIUM.eyebrow}
            </p>
            <h2 id={headingId} ref={heading} tabIndex={-1} data-slot="podium-title" data-bulb="gold" className="neon font-display text-4xl text-fg sm:text-5xl">
              {PODIUM.title}
            </h2>
          </div>
          {animate && (mode === 'play' || mode === 'done') && (
            <button type="button" className="btn-ghost" data-slot="reveal-control" onClick={toggle}>
              {mode === 'play' ? PODIUM.skip : PODIUM.replay}
            </button>
          )}
        </div>
        {sample && <p data-slot="podium-sample">{PODIUM.sampleNote}</p>}

        {mode === 'wait' ? (
          <p data-slot="podium-wait" role="status">
            {loading ? PODIUM.loading : PODIUM.empty}
          </p>

          <ol data-slot="podium-list" data-count={rows.length}>

            ))}
          </ol>
        )}
        {mode !== 'wait' && <p data-slot="podium-rule">{PODIUM.rule}</p>}
      </div>
    </section>
  )
}
````

### D.1: `apps/lights/e2e/flash.ts`

The WCAG 2.3.1 probe (M4 Appendix C.3) as a test helper, with a whole-screen measure and a 256-entry sRGB table so a 1280×800 run takes about a minute.

````ts
// WCAG 2.3.1 flash probe (M4 Appendix C.3, as a helper): steps every animation on the page
// through time with the Web Animations API, screenshots each step and counts flashes.
// A transition is a swing of at least 0.1 in relative luminance with the darker side below
// 0.8; a flash is a pair of opposing transitions. Two measures:
//   - `block`: the worst 6×6 px block, flashes in any 1 s (M4's measure).
//   - `screen`: the whole screen at once. One ramp crosses the threshold in neighboring
//     blocks a frame or two apart, so same-direction transitions under 100 ms apart are one
//     event (something brightens, or darkens); a flash is a pair of opposing events.
// Pure translations (ticker, snow) are held still: they move rather than blink (2.2.2).
import type { Page } from '@playwright/test'

const DT = 50 // ms between samples (20 fps): resolves up to 10 flashes per second
const BLOCK = 6
const MOVERS = ['marquee', 'snowfall-near', 'snowfall-far']

/** Relative luminance per 6×6 block of a screenshot, decoded in a blank page of the same
 *  browser. JPEG at quality 95: several times faster than PNG at 1280×800, and its error
 *  averaged over a 6×6 block is far below the 0.1 threshold. */
async function blocks(decoder: Page, jpeg: Buffer): Promise<number[]> {
  return decoder.evaluate(
    async ({ b64, BLOCK }) => {
      const img = await createImageBitmap(await (await fetch('data:image/jpeg;base64,' + b64)).blob())
      const c = new OffscreenCanvas(img.width, img.height)
      const g = c.getContext('2d')!
      g.drawImage(img, 0, 0)
      const { data, width, height } = g.getImageData(0, 0, img.width, img.height)
      // sRGB → linear, as a table (the pow per pixel is what makes a probe slow).
      const lin = Array.from({ length: 256 }, (_, i) => ((i /= 255) <= 0.04045 ? i / 12.92 : ((i + 0.055) / 1.055) ** 2.4))
      const bw = Math.floor(width / BLOCK)
      const bh = Math.floor(height / BLOCK)
      const out = new Array<number>(bw * bh).fill(0)
      for (let y = 0; y < bh * BLOCK; y++)
        for (let x = 0; x < bw * BLOCK; x++) {
          const i = (y * width + x) * 4
          out[Math.floor(y / BLOCK) * bw + Math.floor(x / BLOCK)] +=
            0.2126 * lin[data[i]] + 0.7152 * lin[data[i + 1]] + 0.0722 * lin[data[i + 2]]
        }
      return out.map((v) => v / (BLOCK * BLOCK))
    },
    { b64: jpeg.toString('base64'), BLOCK },
  )
}

/** Most pairs among `times` (sorted) that fit in any 1-second window. */
function perSecond(times: number[]): number {
  let most = 0
  for (let i = 0; i < times.length; i++) {
    let n = 0
    while (i + n < times.length && times[i + n] < times[i] + 1000) n++
    most = Math.max(most, Math.floor(n / 2))
  }
  return most
}

/** Step the page's animations from 0 to `durationMs` and measure. `only`: step just these
 *  animations and hold the rest at 0 (e.g. the reveal without the hero's ambient loops).
 *  Pause the page's own timers first (page.clock) if something would end the animations. */
export async function flashProbe(
  page: Page,
  durationMs: number,
  only: string[] | null = null,
): Promise<{ block: number; screen: number; stepped: string[] }> {
  const decoder = await page.context().newPage()
  const series: number[][] = []
  const names = new Set<string>()
  for (let t = 0; t <= durationMs; t += DT) {
    const running = await page.evaluate(
      ([t, movers, only]) => {
        const anims = document.getAnimations() as (Animation & { animationName?: string })[]
        const step = (name: string) => !movers.includes(name) && (only === null || only.includes(name))
        for (const a of anims) {
          a.pause()
          a.currentTime = step(a.animationName ?? '') ? t : 0
        }
        return anims.map((a) => a.animationName ?? '').filter(step)
      },
      [t, MOVERS, only] as const,
    )
    running.forEach((n) => n && names.add(n))
    series.push(await blocks(decoder, await page.screenshot({ type: 'jpeg', quality: 95 })))
  }
  await decoder.close()

  let block = 0
  const events: [number, number][] = []
  for (let b = 0; b < series[0].length; b++) {
    const times: number[] = []
    let ref = series[0][b]
    let dir = 0
    for (let k = 1; k < series.length; k++) {
      const L = series[k][b]
      if (dir === 1 && L > ref) ref = L
      else if (dir === -1 && L < ref) ref = L
      const up = L - ref >= 0.1
      const down = ref - L >= 0.1
      if ((up || down) && Math.min(L, ref) < 0.8) {
        dir = up ? 1 : -1
        times.push(k * DT)
        events.push([k * DT, dir])
        ref = L
      }
    }
    block = Math.max(block, perSecond(times))
  }
  const merged: number[] = []
  for (const d of [1, -1]) {
    const ts = [...new Set(events.filter(([, x]) => x === d).map(([t]) => t))].sort((a, b) => a - b)
    ts.forEach((t, i) => (i === 0 || t - ts[i - 1] > 100) && merged.push(t))
  }
  return { block, screen: perSecond(merged.sort((a, b) => a - b)), stepped: [...names].sort() }
}
````

## Appendix F: extraction script (save as `$S/m7/extract.py`; never commit)

### F.1 extract.py

````python
#!/usr/bin/env python3
"""Write the M7 appendices into the repo.

Usage (from the repo root): python3 extract.py docs/milestones/M7.md

Every "### X.n: `path`" heading in Appendices A-E is followed by one fenced block (four
backticks), sometimes after a one-line note. Plain headings write the file, "(append)" appends
the block once, "(diff)" applies it with `git apply` (skipped if already applied). Safe to run
twice.
"""
import os
import re
import subprocess
import sys
import tempfile

text = open(sys.argv[1], encoding='utf-8').read()
pattern = re.compile(
    r'^### ([A-E])\.(\d+): `([^`]+)`( \((append|diff)\))?\n\n(?:[^`\n][^\n]*\n\n)?````[a-z]*\n(.*?)\n````$', re.S | re.M
)
count = 0
for m in pattern.finditer(text):
    path, mode, body = m.group(3), m.group(5), m.group(6) + '\n'
    if mode == 'diff':
        with tempfile.NamedTemporaryFile('w', suffix='.diff', delete=False) as f:
            f.write(body)
        if subprocess.run(['git', 'apply', '--check', f.name], capture_output=True).returncode == 0:
            subprocess.run(['git', 'apply', f.name], check=True)
            print('patched ', path)
        elif subprocess.run(['git', 'apply', '-R', '--check', f.name], capture_output=True).returncode == 0:
            print('ok      ', path, '(already patched)')
        else:
            sys.exit(f'cannot apply the diff for {path}')
        os.unlink(f.name)
    elif mode == 'append':
        current = open(path, encoding='utf-8').read()
        if body.strip() in current:
            print('ok      ', path, '(already appended)')
        else:
            with open(path, 'a', encoding='utf-8') as f:
                f.write(body)
            print('appended', path)
    else:
        os.makedirs(os.path.dirname(path) or '.', exist_ok=True)
        with open(path, 'w', encoding='utf-8') as f:
            f.write(body)
        print('wrote   ', path)
    count += 1
print(count, 'appendix files')
````

````js
async function luma(decoder, png) {
  return decoder.evaluate(async ({ b64, BLOCK }) => {
    const img = await createImageBitmap(await (await fetch('data:image/png;base64,' + b64)).blob())
    const c = new OffscreenCanvas(img.width, img.height), g = c.getContext('2d')
    g.drawImage(img, 0, 0)

    const bw = Math.floor(width / BLOCK), bh = Math.floor(height / BLOCK), out = new Array(bw * bh).fill(0)
    for (let y = 0; y < bh * BLOCK; y++) for (let x = 0; x < bw * BLOCK; x++) {

    }
    return out.map((v) => v / (BLOCK * BLOCK))

function analyse(series, cols) {
  let worst = 0, where = ''
  const all = [] // [time, direction] of every block transition
  for (let b = 0; b < series[0].length; b++) {
    const times = []
    let ref = series[0][b], dir = 0
    for (let k = 1; k < series.length; k++) {
      const L = series[k][b]
      if (dir === 1 && L > ref) ref = L
      else if (dir === -1 && L < ref) ref = L
      if (L - ref >= 0.1 && Math.min(L, ref) < 0.8) { times.push(k * DT); all.push([k * DT, 1]); ref = L; dir = 1 }
      else if (ref - L >= 0.1 && Math.min(L, ref) < 0.8) { times.push(k * DT); all.push([k * DT, -1]); ref = L; dir = -1 }
    }
    for (let i = 0; i < times.length; i++) {
      let n = 0
      while (i + n < times.length && times[i + n] < times[i] + 1000) n++

    screen = Math.max(screen, Math.floor(n / 2))

    if (process.env.DEBUG) console.log('  transition steps (ms):', steps.join(' '))
    await ctx.close()
  }
  await browser.close()
  process.exit(fail ? 1 : 0)
````
