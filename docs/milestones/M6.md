## Goal

Neighbors can find the houses and get there, in the M4 visual language:

- **Lineup ⇄ map.** A two-button switch next to "The Lineup" shows the same approved houses
  as cards or on a dark map. The map view is in the URL (`/?view=map`), survives a reload and
  can be shared; `&house=<n>` opens it on one house.
- **The map** (lazy-loaded MapLibre + CARTO Dark Matter, the module M5 started): every
  approved house is a glowing C9 bulb in a rotating color; the picked one grows and pulses.
  Tapping (or Tab + Enter) a bulb opens the house's photo, number, name, address and a
  **Directions** button: a bottom sheet on phones, a side panel on desktop. A locate button
  shows where you are. Offline, the map says so and offers the lineup.
- **Navigate.** Every card and the sheet have a directions button that opens the platform's
  maps app (plan §7 decision 12): Apple Maps on iPhone and iPad, Google Maps everywhere else,
  routed to the house's coordinates.
- Only **approved** houses appear. An owner whose entry is waiting is told why it isn't there.
- Unit tests for the URLs and the platform detection across real user agents; Playwright
  tests for the toggle, picking, the sheet, the links, keyboard access, reduced motion, the
  locate button and the offline states. The parade and its Vercel build stay byte-identical.

reduced motion and the pause switch stop the pulse and make camera moves instant; the lineup
is the list alternative.

### A.1: `apps/lights/src/lib/view.ts`

````ts
import { useCallback, useState } from 'react'

/** How the home page shows the houses: the lineup (cards) or the map. It lives in the
 *  query string (?view=map), not a path: it is a view of the same lineup, it survives a
 *  reload, and it can be shared. ?house=<entry number> is the house picked on the map
 *  (and, as in M4, the house the lineup pins first). */
export type View = 'lineup' | 'map'

/** Pure: the view in a query string. Anything but view=map is the lineup. */
export function parseView(search: string): View {
  return new URLSearchParams(search).get('view') === 'map' ? 'map' : 'lineup'
}

/** Pure: the entry number in ?house=, or null. */
export function parseHouse(search: string): number | null {
  const raw = new URLSearchParams(search).get('house')?.trim() ?? ''
  return /^[1-9]\d{0,5}$/.test(raw) ? Number(raw) : null
}

/** Pure: the query string for a view and picked house. Keeps every other parameter
 *  (?preview= for the admin, ?signin); the lineup drops both `view` and `house`. */
export function viewSearch(search: string, view: View, house: number | null = null): string {
  const params = new URLSearchParams(search)
  params.delete('view')
  params.delete('house')
  if (view === 'map') {
    params.set('view', 'map')
    if (house !== null) params.set('house', String(house))
  }
  const query = params.toString()
  return query ? `?${query}` : ''
}

/** Rewrite the address bar without a history entry: toggling views and picking houses
 *  shouldn't fill the back button. The hash (#houses) is dropped. */
export function replaceSearch(search: string) {
  window.history.replaceState(window.history.state, '', window.location.pathname + search)
}

/** The current view, read once from the URL; setting it rewrites the URL. */
export function useView(): [View, (view: View) => void] {
  const [view, setViewState] = useState<View>(() => parseView(window.location.search))
  const setView = useCallback((next: View) => {
    replaceSearch(viewSearch(window.location.search, next))
    setViewState(next)
  }, [])
  return [view, setView]
}

/** Bulb color of a house's map marker. Fixed per entry number (a house keeps its color
 *  across visits and polls), rotating through the C9 palette. */
export const MARKER_COLORS = ['red', 'green', 'gold', 'blue', 'magenta', 'white'] as const
export type MarkerColor = (typeof MARKER_COLORS)[number]
export function markerColor(entryNumber: number): MarkerColor {
  const n = Math.abs(Math.trunc(entryNumber))
  return MARKER_COLORS[(n + MARKER_COLORS.length - 1) % MARKER_COLORS.length]
}
````

````ts
import { detectMapsDevice, directionsUrl, type MapsDevice } from '@wohills/core'
import { NAVIGATE } from '../copy'
import type { House } from '../types'

export interface DirectionsLink {
  href: string
  /** "_blank" on desktop (keep the contest open in its tab); none on phones and tablets,
   *  where the OS hands the link to the maps app and the page stays put. */
  target: '_blank' | undefined
  rel: 'noopener noreferrer' | undefined
  /** Accessible name: "Directions to 1221 Candlewood Dr (Google Maps, new tab)". */
  label: string
  /** Tooltip: "Directions in Google Maps". */
  title: string
}

let device: MapsDevice | null = null
/** Detected once per page load (the user agent doesn't change). */
function currentDevice(): MapsDevice {
  device ??= detectMapsDevice()
  return device
}

/** Pure given `d`: the navigate link for a house. Routes to its coordinates; the address
 *  names the link (plan §7 decision 12). */
export function directionsLink(house: Pick<House, 'lat' | 'lng' | 'address'>, d: MapsDevice = currentDevice()): DirectionsLink {
  const app = d.platform === 'apple' ? NAVIGATE.apple : NAVIGATE.google
  const newTab = !d.handheld
  return {
    href: directionsUrl({ lat: house.lat, lng: house.lng, label: house.address }, d),
    target: newTab ? '_blank' : undefined,
    rel: newTab ? 'noopener noreferrer' : undefined,
    label: NAVIGATE.label(house.address, app, newTab),
    title: NAVIGATE.title(app),
  }
}
````

````tsx
export type LocateProblem = 'denied' | 'unavailable' | 'outside'

export interface HouseMapProps {
  /** Approved houses in the lineup's order (also the markers' tab order). */
  houses: House[]
  /** The geofence (pan limit and the fallback view). */

  selectedId: string | null

// A C9 bulb, tip at the bottom (the marker's anchor). Colors come from [data-bulb] in
// map.css: SVG presentation attributes can't read CSS variables (M4 D16).
const BULB_SVG =
  '<svg width="24" height="30" viewBox="3 1 18 21.5" aria-hidden="true" focusable="false">' +
  '<rect class="bulb-socket" x="9" y="1.5" width="6" height="5" rx="1"/>' +

// Room for the 44px bulbs and the controls when the map frames the houses.
const FIT_PADDING = { top: 64, bottom: 40, left: 40, right: 64 }

/** The browse map: every approved house as a glowing bulb you can tap or tab to. Default
 *  export, loaded with React.lazy; it shares one MapLibre chunk with M5's pin map. */
export default function HouseMap({ houses, box, selectedId, getInset, onSelect, onFail, onLocate }: HouseMapProps) {
  const container = useRef<HTMLDivElement>(null)
  const mapRef = useRef<MapLibreMap | null>(null)
  const markers = useRef(new Map<string, { marker: Marker; el: HTMLButtonElement }>())
  // Read the latest props from map and DOM events without re-creating anything.
  const latest = useRef({ houses, box, selectedId, getInset, onSelect, onFail, onLocate })
  useEffect(() => {
    latest.current = { houses, box, selectedId, getInset, onSelect, onFail, onLocate }
  })

  useEffect(() => {
    const { houses, box, selectedId } = latest.current
    const el = container.current!
    const map = createMap({ container: el, box, center: boxCenter(box), zoom: ZOOM.overview, label: MAP.canvas, attribution: 'top-left' })

    const bounds = pointsBox(houses)

        })
        const on = house.id === latest.current.selectedId
        el.setAttribute('aria-pressed', String(on))
        el.toggleAttribute('data-selected', on)
        entry = { marker: new Marker({ element: el, anchor: 'bottom' }).setLngLat([house.lng, house.lat]).addTo(map), el }
        all.set(house.id, entry)
        added.add(el)
      }
      entry.marker.setLngLat([house.lng, house.lat])
      entry.el.dataset.bulb = markerColor(house.entry_number)
      entry.el.dataset.number = String(house.entry_number)
      entry.el.setAttribute('aria-label', MAP.marker(house.entry_number, house.title, house.address))
    }
    // Tab order = the lineup's order. New bulbs were appended; slot each one in before the
    // next house's bulb. Existing bulbs never move, so a focused bulb keeps its focus.
    let next: HTMLButtonElement | null = null
    for (let i = houses.length - 1; i >= 0; i--) {
      const el = all.get(houses[i].id)!.el
      if (added.has(el) && next) next.before(el)
      next = el
    }
  }, [houses])

  // The picked bulb: lit up, on top, and in the part of the map the sheet doesn't cover.
  useEffect(() => {
    for (const [id, { el }] of markers.current) {
      const on = id === selectedId
      el.setAttribute('aria-pressed', String(on))
      el.toggleAttribute('data-selected', on)
    }
    const map = mapRef.current
    const house = houses.find((h) => h.id === selectedId)
    if (!map || !house) return
    const inset = latest.current.getInset()
    const p = map.project([house.lng, house.lat])
    const { clientWidth: w, clientHeight: h } = map.getContainer()
    if (p.x > 48 && p.x < w - 48 && p.y > 64 && p.y < h - inset - 16) return // already in view
    map.easeTo({
      center: [house.lng, house.lat],
      // Shift the house up by half the sheet, so it lands in the middle of what's visible.
      offset: [0, -inset / 2],
      duration: motionReduced() ? 0 : 600,
    })
    // eslint-disable-next-line react-hooks/exhaustive-deps -- move only when the pick changes, not on every poll
  }, [selectedId])

  // h-full, not absolute inset-0: MapLibre's CSS sets position: relative on its container.
  return <div ref={container} data-slot="house-map-canvas" className="h-full w-full" />
}
````

````ts
export type MapsPlatform = 'apple' | 'google'

export interface Destination {
  lat: number
  lng: number
  /** The street address. Directions URLs can't carry it next to coordinates (see
   *  directionsUrl), so apps use it for the link's accessible name instead. */
  label?: string
}

/** What a device opens and how: the maps app, whether its Apple Maps predates the unified
 *  URLs (iOS/iPadOS < 18.4), and whether it is a phone or tablet (where the OS hands the
 *  link to the app) rather than a desktop browser. */
export interface MapsDevice {
  platform: MapsPlatform
  legacyApple: boolean
  handheld: boolean
}

type NavigatorLike = Pick<Navigator, 'userAgent' | 'platform' | 'maxTouchPoints'>

/** First version Apple documents the unified Maps URLs for ("Adopting unified Maps URLs"). */
const UNIFIED_APPLE = [18, 4] as const

const atLeast = ([major, minor]: readonly [number, number], [m, n]: readonly [number, number]) =>
  major > m || (major === m && minor >= n)

/** iPhone/iPad/iPod → Apple Maps. iPadOS asks for desktop sites and reports itself as a Mac
 *  ("Macintosh", platform MacIntel), so a "Mac" with a touch screen is an iPad: real Macs
 *  report maxTouchPoints 0. Everything else (Android, Windows, macOS, Linux) → Google Maps.
 *  The iOS version comes from "OS 17_5" in the user agent (iOS 26 Safari freezes it at
 *  18_6, still new enough), or for an iPad in desktop mode from Safari's "Version/17.5";
 *  an unknown version counts as new. */
export function detectMapsDevice(nav: NavigatorLike = navigator): MapsDevice {
  const ua = nav.userAgent
  const iosUa = /iPhone|iPad|iPod/.test(ua)
  const iPadAsMac = !iosUa && nav.platform === 'MacIntel' && nav.maxTouchPoints > 1
  if (iosUa || iPadAsMac) {
    const m = iosUa ? ua.match(/ OS (\d+)_(\d+)/) : ua.match(/Version\/(\d+)\.(\d+)/)
    const legacyApple = m ? !atLeast([Number(m[1]), Number(m[2])], UNIFIED_APPLE) : false
    return { platform: 'apple', legacyApple, handheld: true }
  }
  return { platform: 'google', legacyApple: false, handheld: /Android/i.test(ua) }
}

/** Apple Maps on iPhone/iPad, Google Maps everywhere else (plan §7 decision 12). */
export function detectMapsPlatform(nav: NavigatorLike = navigator): MapsPlatform {
  return detectMapsDevice(nav).platform
}

/** A turn-by-turn link that routes to the coordinates, which is more reliable than an
 *  address in a new subdivision. Plain https links: the OS hands them to the Maps or Google
 *  Maps app on phones, and they open the web map anywhere else (a maps:// link would fail
 *  outright without the app).
 *  - Apple, iOS 18.4+: https://maps.apple.com/directions?destination=<lat>,<lng>
 *  - Apple, older:     https://maps.apple.com/?daddr=<lat>,<lng>
 *  - Google:           https://www.google.com/maps/dir/?api=1&destination=<lat>,<lng>
 *  None of them can label a coordinate destination (Apple's `q` is a search, not a label),
 *  so `dest.label` is not part of the URL; the maps app shows the spot's street address. */
export function directionsUrl(dest: Destination, device: MapsPlatform | MapsDevice = detectMapsDevice()): string {
  const d = typeof device === 'string' ? { platform: device, legacyApple: false } : device
  const ll = `${dest.lat.toFixed(6)},${dest.lng.toFixed(6)}`
  if (d.platform === 'apple')
    return d.legacyApple ? `https://maps.apple.com/?daddr=${ll}` : `https://maps.apple.com/directions?destination=${ll}`
  return `https://www.google.com/maps/dir/?api=1&destination=${ll}`
}
````

## Appendix F: extraction script (save as `$S/m6/extract.py`; never commit)

### F.1 extract.py

````python
#!/usr/bin/env python3
"""Write the M6 appendices into the repo.

Usage (from the repo root): python3 extract.py docs/milestones/M6.md

Every "### X.n: `path`" heading in Appendices A-E is followed by one fenced block (four
backticks), sometimes after a one-line note. Plain headings write the file, "(append)" appends
the block once, "(diff)" applies it with `git apply` (skipped if already applied). Safe to run
twice.
"""
import os
import re
import subprocess
import sys
import tempfile

text = open(sys.argv[1], encoding='utf-8').read()
pattern = re.compile(
    r'^### ([A-E])\.(\d+): `([^`]+)`( \((append|diff)\))?\n\n(?:[^`\n][^\n]*\n\n)?````[a-z]*\n(.*?)\n````$', re.S | re.M
)
count = 0
for m in pattern.finditer(text):
    path, mode, body = m.group(3), m.group(5), m.group(6) + '\n'
    if mode == 'diff':
        with tempfile.NamedTemporaryFile('w', suffix='.diff', delete=False) as f:
            f.write(body)
        if subprocess.run(['git', 'apply', '--check', f.name], capture_output=True).returncode == 0:
            subprocess.run(['git', 'apply', f.name], check=True)
            print('patched ', path)
        elif subprocess.run(['git', 'apply', '-R', '--check', f.name], capture_output=True).returncode == 0:
            print('ok      ', path, '(already patched)')
        else:
            sys.exit(f'cannot apply the diff for {path}')
        os.unlink(f.name)
    elif mode == 'append':
        current = open(path, encoding='utf-8').read()
        if body.strip() in current:
            print('ok      ', path, '(already appended)')
        else:
            with open(path, 'a', encoding='utf-8') as f:
                f.write(body)
            print('appended', path)
    else:
        os.makedirs(os.path.dirname(path) or '.', exist_ok=True)
        with open(path, 'w', encoding='utf-8') as f:
            f.write(body)
        print('wrote   ', path)
    count += 1
print(count, 'appendix files')
````
